cmake_minimum_required(VERSION 3.10)
project(CFD_FVM_Solver LANGUAGES C)

# 设置C标准和编译选项
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_FLAGS_RELEASE "-O3 -Wall")
set(CMAKE_C_FLAGS_DEBUG "-g -Wall")

# 包含头文件目录
include_directories(include)

# 创建可执行文件
add_executable(fvm_solver src/RP_FVM.c)

# 链接数学库
target_link_libraries(fvm_solver m)

# 自动复制数据文件到构建目录
file(GLOB DATA_FILES "data/*")
if(DATA_FILES)
    add_custom_command(TARGET fvm_solver POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/data
        COMMAND ${CMAKE_COMMAND} -E copy ${DATA_FILES} ${CMAKE_BINARY_DIR}/data
        COMMENT "Copying data files to build directory"
    )
endif()

# 添加运行目标
add_custom_target(run
    COMMAND ${CMAKE_BINARY_DIR}/fvm_solver
    DEPENDS fvm_solver
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Running CFD solver"
)

# 添加清理数据目标
add_custom_target(clean-data
    COMMAND ${CMAKE_COMMAND} -E remove_directory ${CMAKE_BINARY_DIR}/data
    COMMENT "Cleaning data files from build directory"
)
