cmake_minimum_required(VERSION 3.10)
project(CFD_FVM_Solver LANGUAGES C)

# Set C standard and compilation options
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Set default build type
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Find OpenMP
find_package(OpenMP REQUIRED)

# Set compilation flags
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall")

if(CMAKE_BUILD_TYPE STREQUAL "Release")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O3")
elseif(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -g -DDEBUG")
endif()

# Suppress warnings
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wno-unused-variable -Wno-unused-result")

# Include header directories
include_directories(include)

# Create main executable
add_executable(ShockImpact_RPS src/RP_FVM.c)

# Link math library and OpenMP
target_link_libraries(ShockImpact_RPS m OpenMP::OpenMP_C)

# ========== Create simple run script ==========
file(WRITE ${CMAKE_BINARY_DIR}/run.sh
"#!/bin/bash

# Default thread count
THREADS=\${1:-8}

echo \"========================================\"
echo \"CFD Solver Configuration\"
echo \"========================================\"
echo \"Threads: \${THREADS}\"
echo \"Executable: ./ShockImpact_RPS\"
echo \"Working directory: \$(pwd)\"
echo \"Time: \$(date '+%Y-%m-%d %H:%M:%S')\"
echo \"========================================\"

# Set OpenMP thread count
export OMP_NUM_THREADS=\${THREADS}
export OMP_PROC_BIND=true

# Run the program
time ./ShockImpact_RPS

EXIT_CODE=\$?
echo \"========================================\"
echo \"Exit code: \${EXIT_CODE}\"
exit \${EXIT_CODE}
")

# Set script executable permissions
file(COPY ${CMAKE_BINARY_DIR}/run.sh
     DESTINATION ${CMAKE_BINARY_DIR}
     FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)

# ========== Create run targets ==========

# Default run (using script, 8 threads)
add_custom_target(run
    COMMAND bash ./run.sh 8
    DEPENDS ShockImpact_RPS
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Run CFD solver (8 threads)"
)

# Direct run (default thread count)
add_custom_target(run_direct
    COMMAND ./ShockImpact_RPS
    DEPENDS ShockImpact_RPS
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Run solver directly"
)

# Auto-detect core count
add_custom_target(run_auto
    COMMAND bash -c "THREADS=\$$(nproc 2>/dev/null || echo 8) && bash ./run.sh \$$THREADS"
    DEPENDS ShockImpact_RPS
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Run with auto-detected core count"
)

# ========== OpenMP information display target ==========
add_custom_target(openmp_info
    COMMAND ${CMAKE_COMMAND} -E echo "=== OpenMP Configuration Information ===";
    COMMAND ${CMAKE_COMMAND} -E echo "OpenMP version: ${OpenMP_C_VERSION}"
    COMMAND ${CMAKE_COMMAND} -E echo "OpenMP specification date: ${OpenMP_C_SPEC_DATE}"
    COMMAND ${CMAKE_COMMAND} -E echo "Compiler flags: ${OpenMP_C_FLAGS}"
    COMMAND ${CMAKE_COMMAND} -E echo "OpenMP enabled: ${OpenMP_C_FOUND}"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "To test OpenMP:"
    COMMAND ${CMAKE_COMMAND} -E echo "  OMP_NUM_THREADS=2 ./ShockImpact_RPS"
    DEPENDS ShockImpact_RPS
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Display OpenMP version and configuration information"
    VERBATIM
)

# ========== OpenMP verification target ==========
add_custom_target(verify_openmp
    COMMAND ${CMAKE_COMMAND} -E echo "=== System Core Information ==="
    COMMAND ${CMAKE_COMMAND} -E echo -n "Physical cores: "
    COMMAND bash -c "nproc 2>/dev/null || echo 8"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "=== OpenMP Information ==="
    COMMAND ${CMAKE_COMMAND} -E echo "OpenMP version: ${OpenMP_C_VERSION}"
    COMMAND ${CMAKE_COMMAND} -E echo "Compiler flags: ${OpenMP_C_FLAGS}"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "=== Recommended Run Parameters ==="
    COMMAND bash -c "echo 'Suggested thread count: \$$(nproc 2>/dev/null || echo 8)'"
    COMMAND bash -c "echo 'Run example: OMP_NUM_THREADS=\$$(nproc 2>/dev/null || echo 8) ./ShockImpact_RPS'"
    DEPENDS ShockImpact_RPS
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Verify OpenMP configuration and system core information"
    VERBATIM
)

# ========== Display help information ==========
add_custom_target(usage
    COMMAND ${CMAKE_COMMAND} -E echo "=== CFD_FVM_Solver Usage Help ==="
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "Build commands:"
    COMMAND ${CMAKE_COMMAND} -E echo "  make                        # Build program"
    COMMAND ${CMAKE_COMMAND} -E echo "  make clean                  # Clean build files"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "Run commands:"
    COMMAND ${CMAKE_COMMAND} -E echo "  make run                    # Run with 8 threads"
    COMMAND ${CMAKE_COMMAND} -E echo "  make run_direct             # Run directly (default thread count)"
    COMMAND ${CMAKE_COMMAND} -E echo "  make run_auto               # Run with auto-detected core count"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "Information commands:"
    COMMAND ${CMAKE_COMMAND} -E echo "  make openmp_info            # Display OpenMP version information"
    COMMAND ${CMAKE_COMMAND} -E echo "  make verify_openmp          # Verify OpenMP configuration"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "Direct execution:"
    COMMAND ${CMAKE_COMMAND} -E echo "  ./ShockImpact_RPS           # Run directly (using default thread count)"
    COMMAND ${CMAKE_COMMAND} -E echo "  bash ./run.sh [threads]     # Use run script"
    COMMAND ${CMAKE_COMMAND} -E echo ""
    COMMAND ${CMAKE_COMMAND} -E echo "Set environment variables:"
    COMMAND ${CMAKE_COMMAND} -E echo "  export OMP_NUM_THREADS=4    # Set thread count to 4"
    DEPENDS ShockImpact_RPS
    WORKING_DIRECTORY ${CMAKE_BINARY_DIR}
    COMMENT "Display usage help"
    VERBATIM
)

# ========== Configuration summary ==========
message(STATUS "")
message(STATUS "================================================")
message(STATUS "CFD_FVM_Solver Configuration Complete")
message(STATUS "================================================")
message(STATUS "Build type:        ${CMAKE_BUILD_TYPE}")
message(STATUS "Compiler:          ${CMAKE_C_COMPILER}")
message(STATUS "Compilation flags: ${CMAKE_C_FLAGS}")
if(OpenMP_C_FOUND)
    message(STATUS "OpenMP support:    ✅ Enabled (version: ${OpenMP_C_VERSION})")
else()
    message(STATUS "OpenMP support:    ❌ Disabled")
endif()
message(STATUS "")
message(STATUS "Available commands:")
message(STATUS "  make                    # Build program")
message(STATUS "  make run               # Run solver (8 threads)")
message(STATUS "  make run_auto          # Run with auto-detected core count")
message(STATUS "  make openmp_info       # Display OpenMP information")
message(STATUS "  make verify_openmp     # Verify OpenMP configuration")
message(STATUS "  make usage             # Display help")
message(STATUS "")
message(STATUS "Direct execution:")
message(STATUS "  ./ShockImpact_RPS      # Using default thread count")
message(STATUS "  bash ./run.sh [threads] # Using run script")
message(STATUS "================================================")
